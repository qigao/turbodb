option(SQLPARSER_SQLITE_ENABLE_UPDATE_DELETE_LIMIT
  "Enable SQLite's optional UPDATE/DELETE ORDER BY and LIMIT syntax" OFF)
set(SQLPARSER_SQLITE_GRAMMAR_DEFINES SQLITE)
if(SQLPARSER_SQLITE_ENABLE_UPDATE_DELETE_LIMIT)
  list(APPEND SQLPARSER_SQLITE_GRAMMAR_DEFINES SQLITE_ENABLE_UPDATE_DELETE_LIMIT)
endif()

cmake_add_grammar(SQLPARSER
  LEXER_RE ${CMAKE_CURRENT_SOURCE_DIR}/src/lexer.re
  GRAMMAR_Y ${CMAKE_CURRENT_SOURCE_DIR}/src/grammar.y
  RE2C_EXECUTABLE "${RE2C_EXECUTABLE}"
  LEMON_EXECUTABLE "${LEMON_EXECUTABLE}"
  LEMON_TEMPLATE "${LEMPAR}"
  GRAMMAR_DEPENDS ${LEMON_DEPENDS} src/internal.h
  LEXER_DEPENDS src/internal.h
  LEXER_DEPENDS_ON_GRAMMAR FOLDER sqlparser)

cmake_add_grammar(SQLPARSER_SQLITE
  GRAMMAR_Y ${CMAKE_CURRENT_SOURCE_DIR}/src/grammar.y
  LEMON_EXECUTABLE "${LEMON_EXECUTABLE}" LEMON_TEMPLATE "${LEMPAR}"
  GRAMMAR_DEFINES ${SQLPARSER_SQLITE_GRAMMAR_DEFINES}
  GRAMMAR_DEPENDS ${LEMON_DEPENDS} src/internal.h
  FOLDER sqlparser)

add_library(sqlparser STATIC src/parser.c src/ast.c
  ${SQLPARSER_LEXER_GEN} ${SQLPARSER_GRAMMAR_GEN} ${SQLPARSER_SQLITE_GRAMMAR_GEN})
cmake_config_target(sqlparser ALIAS TurboDB::SqlParser FOLDER sqlparser
  EXPORT_SET TurboDBTargets EXPORT_NAME SqlParser)
target_include_directories(sqlparser
  PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
         $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
  PRIVATE src ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(sqlparser PRIVATE Salts::CSTL Salts::Core)
target_compile_definitions(sqlparser PUBLIC
  SQLPARSER_SQLITE_ENABLE_UPDATE_DELETE_LIMIT=$<BOOL:${SQLPARSER_SQLITE_ENABLE_UPDATE_DELETE_LIMIT}>)
cmake_install_headers(FILES include/sqlparser/sqlparser.h
  DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/sqlparser)
if(BUILD_TESTS)
  add_subdirectory(tests)
endif()
if(BUILD_EXAMPLES)
  add_subdirectory(examples)
endif()
